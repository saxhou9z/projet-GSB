Sub Create_Tables()

DoCmd.RunSQL "CREATE TABLE Salarie(" & _
   "id_salarie VARCHAR(50)," & _
   "Nom TEXT," & _
   "Prenom TEXT," & _
   "mot_de_passe TEXT," & _
   "Age INT," & _
   "Adresse TEXT," & _
   "type_utilisateur TEXT," & _
   "PRIMARY KEY(id_salarie)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Responsable_Secteur(" & _
   "id_salarie VARCHAR(50)," & _
   "PRIMARY KEY(id_salarie)," & _
   "FOREIGN KEY(id_salarie) REFERENCES Salarie(id_salarie)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Secteur(" & _
   "id_secteur VARCHAR(50)," & _
   "libelle_secteur TEXT," & _
   "PRIMARY KEY(id_secteur)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Region(" & _
   "id_region VARCHAR(50)," & _
   "libelle_region TEXT," & _
   "PRIMARY KEY(id_region)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Délégue_regional(" & _
   "id_salarie VARCHAR(50)," & _
   "PRIMARY KEY(id_salarie)," & _
   "FOREIGN KEY(id_salarie) REFERENCES Salarie(id_salarie)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Visiteur(" & _
   "id_salarie VARCHAR(50)," & _
   "PRIMARY KEY(id_salarie)," & _
   "FOREIGN KEY(id_salarie) REFERENCES Salarie(id_salarie)" & _
");"   

DoCmd.RunSQL "CREATE TABLE date_visite(" & _
   "date_du_jour DATE," & _
   "heure_rdv TIME," & _
   "PRIMARY KEY(date_du_jour)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Professionneles_de_santés(" & _
   "id_pds VARCHAR(50)," & _
   "nom_pds TEXT," & _
   "prenom_pds TEXT," & _
   "age_pds INT," & _
   "metier TEXT," & _
   "adresses_pds TEXT," & _
   "CP_pds VARCHAR(5)," & _
   "Ville_pds TEXT," & _
   "PRIMARY KEY(id_pds)" & _
");"   

DoCmd.RunSQL "CREATE TABLE famille(" & _
   "codecat VARCHAR(50)," & _
   "libellecat VARCHAR(50)," & _
   "PRIMARY KEY(codecat)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Medicaments(" & _
   "id_medicaments VARCHAR(50)," & _
   "designation TEXT," & _
   "prix DECIMAL(15,2)," & _
   "codecat VARCHAR(50) NOT NULL," & _
   "PRIMARY KEY(id_medicaments)," & _
   "FOREIGN KEY(codecat) REFERENCES famille(codecat)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Gère(" & _
   "id_salarie VARCHAR(50)," & _
   "id_secteur VARCHAR(50)," & _
   "PRIMARY KEY(id_salarie, id_secteur)," & _
   "FOREIGN KEY(id_salarie) REFERENCES Responsable_Secteur(id_salarie)," & _
   "FOREIGN KEY(id_secteur) REFERENCES Secteur(id_secteur)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Composer(" & _
   "id_secteur VARCHAR(50)," & _
   "id_region VARCHAR(50)," & _
   "PRIMARY KEY(id_secteur, id_region)," & _
   "FOREIGN KEY(id_secteur) REFERENCES Secteur(id_secteur)," & _
   "FOREIGN KEY(id_region) REFERENCES Region(id_region)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Diriger(" & _
   "id_region VARCHAR(50)," & _
   "id_salarie VARCHAR(50)," & _
   "PRIMARY KEY(id_region, id_salarie)," & _
   "FOREIGN KEY(id_region) REFERENCES Region(id_region)," & _
   "FOREIGN KEY(id_salarie) REFERENCES Délégue_regional(id_salarie)" & _
");"   

DoCmd.RunSQL "CREATE TABLE etre_responsable(" & _
   "id_salarie VARCHAR(50)," & _
   "id_salarie_1 VARCHAR(50)," & _
   "PRIMARY KEY(id_salarie, id_salarie_1)," & _
   "FOREIGN KEY(id_salarie) REFERENCES Délégue_regional(id_salarie)," & _
   "FOREIGN KEY(id_salarie_1) REFERENCES Visiteur(id_salarie)" & _
");"   

DoCmd.RunSQL "CREATE TABLE Visiter(" & _
   "id_salarie VARCHAR(50)," & _
   "date_du_jour DATE," & _
   "id_medicaments VARCHAR(50)," & _
   "id_pds VARCHAR(50)," & _
   "Commentaire TEXT," & _
   "PRIMARY KEY(id_salarie, date_du_jour, id_medicaments, id_pds)," & _
   "FOREIGN KEY(id_salarie) REFERENCES Visiteur(id_salarie)," & _
   "FOREIGN KEY(date_du_jour) REFERENCES date_visite(date_du_jour)," & _
   "FOREIGN KEY(id_medicaments) REFERENCES Medicaments(id_medicaments)," & _
   "FOREIGN KEY(id_pds) REFERENCES Professionneles_de_santés(id_pds)" & _
");"   

End Sub
